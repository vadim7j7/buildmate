# Python Backend Code Patterns (Flask)

Reference patterns for the Flask service layer. Read with `patterns/flask-patterns.md`
(app factory, blueprints, models, config, tests). Everything here is **synchronous**:
Flask-SQLAlchemy's `db.session` is a sync, app-context-scoped session.

---

## 1. Schema Pattern (Pydantic v2)

Schemas define request/response shapes using Pydantic v2.

```python
# src/app/schemas/project.py
from datetime import datetime

from pydantic import BaseModel, ConfigDict, Field


class ProjectBase(BaseModel):
    """Shared fields for project schemas."""

    name: str = Field(min_length=1, max_length=255)
    description: str | None = None


class ProjectCreate(ProjectBase):
    """Schema for creating a project."""


class ProjectUpdate(BaseModel):
    """Schema for updating a project (all fields optional)."""

    name: str | None = Field(default=None, min_length=1, max_length=255)
    description: str | None = None


class ProjectRead(ProjectBase):
    """Schema for reading a project (includes DB fields)."""

    model_config = ConfigDict(from_attributes=True)

    id: int
    created_at: datetime
    updated_at: datetime
```

### Schema Rules

- Base schema holds shared required fields; Create inherits from Base
- Update has all optional fields (for partial updates)
- Read adds `ConfigDict(from_attributes=True)` to validate ORM objects
- Serialize for `jsonify()` with `model_dump(mode="json")` (datetimes become strings)
- Use `model_validate()` to parse (never `.parse_obj()`); never `.dict()`

## 2. Service Pattern (sync, Flask-SQLAlchemy)

Services hold business logic and use `db.session` directly (it is scoped to the
current app context / request).

```python
# src/app/services/project_service.py
from sqlalchemy import select
from sqlalchemy.orm import selectinload

from app.extensions import db
from app.models.project import Project
from app.schemas.project import ProjectCreate, ProjectUpdate


class ProjectService:
    """Business logic for project operations."""

    def list(self, *, skip: int = 0, limit: int = 20) -> list[Project]:
        """List projects with pagination."""
        stmt = (
            select(Project)
            .options(selectinload(Project.owner))
            .order_by(Project.id)
            .offset(skip)
            .limit(limit)
        )
        return list(db.session.execute(stmt).scalars())

    def get_by_id(self, project_id: int) -> Project | None:
        """Get a project by primary key."""
        return db.session.get(Project, project_id)

    def create(self, payload: ProjectCreate, *, owner_id: int) -> Project:
        """Create a new project owned by ``owner_id``."""
        project = Project(**payload.model_dump(), owner_id=owner_id)
        db.session.add(project)
        db.session.commit()
        return project

    def update(self, project_id: int, payload: ProjectUpdate) -> Project | None:
        """Update an existing project."""
        project = self.get_by_id(project_id)
        if project is None:
            return None
        for field, value in payload.model_dump(exclude_unset=True).items():
            setattr(project, field, value)
        db.session.commit()
        return project

    def delete(self, project_id: int) -> bool:
        """Delete a project by ID."""
        project = self.get_by_id(project_id)
        if project is None:
            return False
        db.session.delete(project)
        db.session.commit()
        return True
```

### Service Rules

- Plain `def` methods; never `async def`, `await` or `async with db.session()`
- Query with `db.session.execute(select(...))`, `db.session.get(...)`; never `Model.query`
- Eager-load relationships the caller uses (`selectinload()` / `joinedload()`)
- One `commit()` per unit of work; on error the global 500 handler rolls back
- Use keyword-only args for optional parameters
- Return `None`/`False` for not-found (let the view decide the status code)
- Add docstrings on all public methods

## 3. Migration Pattern (Flask-Migrate)

Flask-Migrate wraps Alembic: `flask db migrate -m "..."`, `flask db upgrade`,
`flask db downgrade`. Review every autogenerated file.

```python
"""create projects table"""

import sqlalchemy as sa
from alembic import op


def upgrade() -> None:
    op.create_table(
        "projects",
        sa.Column("id", sa.Integer(), primary_key=True),
        sa.Column("name", sa.String(255), nullable=False),
        sa.Column("description", sa.Text(), nullable=True),
        sa.Column("owner_id", sa.Integer(), sa.ForeignKey("users.id"), nullable=False),
        sa.Column("created_at", sa.DateTime(), server_default=sa.func.now(), nullable=False),
        sa.Column("updated_at", sa.DateTime(), server_default=sa.func.now(), nullable=False),
    )
    op.create_index("ix_projects_owner_id", "projects", ["owner_id"])


def downgrade() -> None:
    op.drop_index("ix_projects_owner_id", table_name="projects")
    op.drop_table("projects")
```

### Migration Rules

- Always include `upgrade()` and `downgrade()`; verify the downgrade runs
- Keep migrations atomic (one logical change per migration)
- Name indexes and constraints explicitly
- Use `op.batch_alter_table()` when altering tables on SQLite

## 4. Service Tests

```python
# tests/services/test_project_service.py
import pytest
from flask import Flask

from app.schemas.project import ProjectCreate
from app.services.project_service import ProjectService


class TestProjectService:
    """Tests for ProjectService business logic (uses the `app` fixture from conftest)."""

    def test_create_project(self, app: Flask, user_factory) -> None:
        owner = user_factory()
        project = ProjectService().create(ProjectCreate(name="Test"), owner_id=owner.id)

        assert project.id is not None
        assert project.name == "Test"

    def test_get_by_id_returns_none_for_missing(self, app: Flask) -> None:
        assert ProjectService().get_by_id(99999) is None

    @pytest.mark.parametrize("name", ["", "x" * 256])
    def test_create_schema_rejects_bad_names(self, name: str) -> None:
        with pytest.raises(ValueError):
            ProjectCreate(name=name)
```

### Test Rules

- Tests are sync; run DB work inside the `app` fixture's app context
- Use `pytest.raises` for expected exceptions and `pytest.mark.parametrize` for inputs
- Isolate tests with per-function database state

## 5. Error Handling and Logging

```python
import logging

logger = logging.getLogger(__name__)


class ServiceError(Exception):
    """Base exception for service-layer errors."""


class NotFoundError(ServiceError):
    """Raised when a requested resource does not exist."""


class ConflictError(ServiceError):
    """Raised on duplicate or conflicting state."""
```

- Define domain exceptions inheriting from `ServiceError`; map them to HTTP codes in
  `register_error_handlers()`
- Catch specific exceptions (never bare `except`); `logger.exception()` in except blocks
- Use `%`-style formatting in log calls; never log secrets or PII
